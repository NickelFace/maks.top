{{/* A source file: path in the header, line numbers on, one line highlightable. {{< file path="/etc/systemd/system/ssh.socket.d/override.conf" lang="ini" hl="4" >}} [Socket] ListenStream= {{< /file >}} path required. The reason this block type exists — a config snippet with no path is the thing the site used to ship. lang Chroma lexer name. Default: guessed from the path extension. hl comma-separated line numbers to highlight (Chroma hl_lines). */}} {{- $path := .Get "path" -}} {{- $lang := .Get "lang" -}} {{- if not $lang -}} {{- $ext := path.Ext $path -}} {{- $map := dict ".conf" "ini" ".ini" "ini" ".yml" "yaml" ".yaml" "yaml" ".json" "json" ".sh" "bash" ".py" "python" ".go" "go" ".js" "javascript" ".css" "css" ".html" "html" ".toml" "toml" ".service" "ini" ".socket" "ini" -}} {{- $lang = index $map $ext | default "text" -}} {{- end -}} {{- $opts := "lineNos=table,lineNumbersInTable=true,tabWidth=2" -}} {{- with .Get "hl" }}{{ $opts = printf "%s,hl_lines=%s" $opts . }}{{ end -}}
{{ $path }} {{ $lang }}
{{ highlight (trim .Inner "\n") $lang $opts }}
